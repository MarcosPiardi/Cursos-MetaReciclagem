# Arquivo: nginx/nginx.conf
# Caminho: eventosmeta/nginx/nginx.conf
# Finalidade: Configuracao principal do Nginx (Producao)
# Atualizacoes:
#  - 06/08/2026 - Adicionado server_tokens off (oculta versao do nginx)
#               - Adicionados tipos ao gzip (application/javascript,
#                 application/xml, image/svg+xml)
# =============================================================

user nginx;
worker_processes auto;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
}

http {
    include       /etc/nginx/mime.types;
    default_type  application/octet-stream;

    # Oculta a versao do nginx nas respostas (seguranca)
    server_tokens off;

    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                     '$status $body_bytes_sent "$http_referer" '
                     '"$http_user_agent" "$http_x_forwarded_for"';

    access_log /var/log/nginx/access.log main;
    error_log  /var/log/nginx/error.log;

    sendfile        on;
    tcp_nopush      on;
    keepalive_timeout 65;

    gzip on;
    gzip_types text/plain text/css text/javascript application/json
               application/javascript application/xml image/svg+xml;
    gzip_min_length 1000;

    include /etc/nginx/conf.d/*.conf;
}

set -e

# Aguardar PostgreSQL estar pronto antes de continuar
echo "Aguardando PostgreSQL..."
while ! python -c "
import socket, os
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
host = os.environ.get('DATABASE_HOST', 'db_eventosmeta')

    s.connect((host, port))
    s.close()
    exit(0)
except Exception:
    exit(1)

    sleep 2
done

python manage.py migrate --noinput

# Coletar arquivos estaticos apenas em producao

# Arquivo: nginx.conf
# Caminho: eventosmeta/nginx.conf
# Finalidade: Configuração do Nginx (proxy reverso + estáticos) - Desenvolvimento
# Atualizações:
#  - 01/08/2026 - Adicionadas diretivas proxy_http_version e
#                 Connection "" para o keepalive do upstream funcionar
# ============================================================

upstream web {
    server web_eventosmeta:8000;
    keepalive 64;
}

server {
    listen 80;
    # server_name 10.30.166.21 localhost 127.0.0.1;
    server_name _;

    client_max_body_size 100M;

    gzip on;
    gzip_types text/plain text/css text/javascript application/json;
    gzip_min_length 1000;

    location /static/ {
        alias /app/staticfiles_collected/;
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    location /media/ {
        alias /app/media/;
        expires 7d;
        add_header Cache-Control "public";
    }

    location /health/ {
        access_log off;
        return 200 "ok";
    }

    location / {
        proxy_pass http://web;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }
}
