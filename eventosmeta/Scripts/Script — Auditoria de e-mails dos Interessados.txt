Script — Auditoria de e-mails dos Interessados

python -c "
import django, os
os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'config.settings')
django.setup()

from apps.interessados.models import Interessado
from django.db.models import Count

print('=== SEM E-MAIL ===')
sem_email = Interessado.objects.filter(email='')
for i in sem_email:
    print('  ' + i.nome + ' | CPF: ' + i.cpf)
print('Total: ' + str(sem_email.count()))

print()
print('=== E-MAILS DUPLICADOS ===')
duplicados = (
    Interessado.objects
    .exclude(email='')
    .values('email')
    .annotate(total=Count('id'))
    .filter(total__gt=1)
)
for d in duplicados:
    nomes = list(Interessado.objects.filter(email=d['email']).values_list('nome', flat=True))
    print('  ' + d['email'] + ' -> ' + str(nomes))
print('Total de e-mails duplicados: ' + str(duplicados.count()))
"
