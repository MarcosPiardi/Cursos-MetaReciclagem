    @pytest.mark.django_db
    def test_data_formatada_methods(self):
        from django.utils import timezone
        import datetime
        evento = EventoFactory(
            data_inicio=datetime.date(2025, 6, 1),
            data_fim=datetime.date(2025, 6, 5),
            horario_inicio=datetime.time(8, 0),
            horario_fim=datetime.time(17, 0),
        )
        admin = EventoAdmin(Evento, AdminSite())
        # Testa data_inicio_formatada
        result_inicio = admin.data_inicio_formatada(evento)
        assert "01/06/2025" in result_inicio
        # Testa data_fim_formatada
        result_fim = admin.data_fim_formatada(evento)
        assert "05/06/2025" in result_fim
        # Testa horario_inicio_formatado
        result_horario_inicio = admin.horario_inicio_formatado(evento)
        assert "08:00" in result_horario_inicio
        # Testa horario_fim_formatado
        result_horario_fim = admin.horario_fim_formatado(evento)
        assert "17:00" in result_horario_fim


class TestEventoAdminActions:
    @pytest.mark.django_db
    def test_classificar_inscricoes_success(self):
        evento = EventoFactory()
        criterio = CriterioFactory(ativo=True)
        EventoCriterioFactory(evento=evento, criterio=criterio, pontos_max=10)
        inter = InteressadoFactory()
        insc = InscricaoFactory(evento=evento, interessado=inter, status="inscrito")
        # Cria request com messagens
        request = HttpRequest()
        request._messages = FallbackStorage(request)
        admin = EventoAdmin(Evento, AdminSite())
        queryset = Evento.objects.filter(pk=evento.pk)
        admin.classificar_inscricoes(request, queryset)
        # Verifica se mensagem de sucesso
        messages = [m.message for m in request._messages]
        assert any("classificadas" in msg for msg in messages)

    @pytest.mark.django_db
    def test_classificar_inscricoes_no_criteria_warning(self):
        evento = EventoFactory()
        request = HttpRequest()
        request._messages = FallbackStorage(request)
        admin = EventoAdmin(Evento, AdminSite())

        admin.classificar_inscricoes(request, queryset)
        messages = [m.message for m in request._messages]
        assert any("critério" in msg.lower() for msg in messages)

        InscricaoFactory(evento=evento, interessado=inter, status="inscrito")
        # Cria request

    @pytest.mark.django_db
    def test_exportar_classificacao_excel_success(self):
        evento = EventoFactory()
        criterio = CriterioFactory(ativo=True)
        EventoCriterioFactory(evento=evento, criterio=criterio, pontos_max=10)
        inter = InteressadoFactory()
        InscricaoFactory(evento=evento, interessado=inter, status="classificado")
        request = HttpRequest()
        admin = EventoAdmin(Evento, AdminSite())
        queryset = Evento.objects.filter(pk=evento.pk)
        response = admin.exportar_classificacao_excel(request, queryset)
        assert response.status_code == 200
        assert response.get("Content-Type") == "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"

    @pytest.mark.django_db
    def test_exportar_classificacao_excel_with_criteria(self):
        evento = EventoFactory()
        criterio1 = CriterioFactory(ativo=True, nome="Critério A")
        criterio2 = CriterioFactory(ativo=True, nome="Critério B")
        EventoCriterioFactory(evento=evento, criterio=criterio1, pontos_max=10)
        EventoCriterioFactory(evento=evento, criterio=criterio2, pontos_max=5)
        inter = InteressadoFactory()
        InscricaoFactory(evento=evento, interessado=inter, status="classificado")
        request = HttpRequest()
        admin = EventoAdmin(Evento, AdminSite())
        queryset = Evento.objects.filter(pk=evento.pk)
        response = admin.exportar_classificacao_excel(request, queryset)
        assert response.status_code == 200
        content = response.getvalue().decode("latin-1")
        assert "Critério A" in content or "Critério" in content
