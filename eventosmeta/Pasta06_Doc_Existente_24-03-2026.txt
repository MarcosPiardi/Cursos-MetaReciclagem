system:
  name: Sistema Acadêmico / MetaReciclagem
  description: >
    Sistema desenvolvido em Django para gestão de eventos,
    inscrições, classificação de candidatos e processos acadêmicos
    posteriores (matrícula, acompanhamento, relatórios).

documentation_strategy:
  approach: referenciada
  rationale: >
    A documentação principal mantém visão sistêmica e arquitetural.
    Subsistemas complexos possuem documentação técnica especializada
    em arquivos separados.

overview:
  main_domains:
    - Eventos
    - Inscrições
    - Classificação
    - Matrícula
  primary_flow: Inscrição -> Classificação -> Resultado -> Matrícula

architecture:
  framework: Django
  language: Python
  patterns:
    - Apps desacoplados
    - Services para regras de negócio
    - Admin Django como interface operacional
    - Templates para apresentação
  separation_of_concerns:
    models: Persistência de dados
    services: Regras de negócio e cálculos
    admin: Operações em massa, exportações e controle
    templates: Interface e UX

apps:
  eventos:
    responsibility:
      - Configuração de eventos
      - Associação de critérios
    notes: Fonte de definição das regras de classificação
  selecao:
    responsibility:
      - Inscrições
      - Classificação
      - Resultado final
    notes: Contém o núcleo de lógica de classificação
  interessados:
    responsibility:
      - Cadastro de candidatos
      - Dados pessoais e critérios sociais
  dashboard:
    responsibility:
      - Visualização administrativa
      - Apoio à tomada de decisão

business_flows:
  classificacao:
    description: >
      Processo de avaliação e ordenação de inscrições com base
      em critérios configuráveis por evento.
    detailed_documentation:
      file: 2026-03-06_resumo_tecnico_classificacao.md
      role: Fonte técnica oficial do subsistema de Classificação
    summary_steps:
      - Leitura dos critérios do evento
      - Cálculo de pontuação
      - Aplicação de ordenação
      - Definição de posição final
      - Persistência do resultado

rules:
  classification:
    source_of_truth:
      - Código (ClassificadorService)
      - Documento técnico especializado (.md)
    key_concepts:
      - Critérios de pontuação vs ordenação
      - Prioridade configurável
      - Desempate por data de inscrição

onboarding:
  recommended_reading_order:
    - documentacao_sistema.yaml
    - 2026-03-06_resumo_tecnico_classificacao.md
    - Código (services -> models -> admin)
  warnings:
    - Não alterar services sem entender critérios
    - Classificação é área sensível do sistema

risks_and_notes:
  - Subsistema de Classificação é crítico e complexo
  - Documentação especializada deve ser mantida alinhada ao código
  - Alterações em critérios impactam diretamente resultados
