<script>
document.addEventListener('DOMContentLoaded', function() {
    // ==========================================
    // TOGGLE DA COLUNA DE FILTROS (DIREITA)
    // ==========================================
    var filterColumn = document.getElementById('changelist-filter');
    
    if (filterColumn) {
        // Cria wrapper para o toggle
        var wrapper = document.createElement('div');
        wrapper.id = 'changelist-filter-wrapper';
        filterColumn.parentNode.insertBefore(wrapper, filterColumn);
        wrapper.appendChild(filterColumn);
        
        // Cria o botão toggle
        var toggleBtn = document.createElement('div');
        toggleBtn.id = 'changelist-filter-toggle';
        toggleBtn.title = 'Mostrar/Ocultar Filtros';
        
        var toggleIcon = document.createElement('span');
        toggleIcon.id = 'changelist-filter-toggle-icon';
        toggleIcon.innerHTML = '»';
        
        toggleBtn.appendChild(toggleIcon);
        wrapper.appendChild(toggleBtn);
        
        // Verifica estado salvo dos filtros
        var filtersHidden = localStorage.getItem('adminFiltersHidden') === 'true';
        if (filtersHidden) {
            document.body.classList.add('filters-hidden');
        }
        
        // Toggle dos filtros ao clicar
        toggleBtn.addEventListener('click', function() {
            document.body.classList.toggle('filters-hidden');
            var isHidden = document.body.classList.contains('filters-hidden');
            
            // Salva estado
            localStorage.setItem('adminFiltersHidden', isHidden);
        });
