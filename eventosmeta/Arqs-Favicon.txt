        function buscarCEP(cep) {
            const loading = document.querySelector('.cep-loading');
            const enderecoInput = document.getElementById('id_endereco_residencial');
            const bairroInput = document.getElementById('id_bairro');
            const cidadeInput = document.getElementById('id_cidade_residencia');
            const ufInput = document.getElementById('id_uf_residencia');
            
            if (loading) loading.style.display = 'block';
            
            fetch(`https://viacep.com.br/ws/${cep}/json/`)
                .then(response => response.json())
                .then(data => {
                    if (loading) loading.style.display = 'none';
                    
                    if (data.erro) {
                        alert('CEP não encontrado!');
                        limparEndereco();
                    } else {
                        if (enderecoInput) {
                            enderecoInput.value = data.logradouro || '';
                            enderecoInput.removeAttribute('readonly');
                        }
                        if (bairroInput) {
                            bairroInput.value = data.bairro || '';
                            bairroInput.removeAttribute('readonly');
                        }
                        if (cidadeInput) {
                            cidadeInput.value = data.localidade || '';
                            cidadeInput.removeAttribute('readonly');
                        }
                        if (ufInput) {
                            ufInput.value = data.uf || '';
                            ufInput.removeAttribute('readonly');
                        }
                        
                        const numEnderecoInput = document.getElementById('id_num_endereco');
                        if (numEnderecoInput) numEnderecoInput.focus();
                    }
                })
                .catch(error => {
                    if (loading) loading.style.display = 'none';
                    console.error('Erro ao buscar CEP:', error);
                    alert('Erro ao buscar CEP. Tente novamente.');
                });
        }
        
        function limparEndereco() {
            const enderecoInput = document.getElementById('id_endereco_residencial');
            const bairroInput = document.getElementById('id_bairro');
            const cidadeInput = document.getElementById('id_cidade_residencia');
            const ufInput = document.getElementById('id_uf_residencia');
            
            if (enderecoInput) enderecoInput.value = '';
            if (bairroInput) bairroInput.value = '';
            if (cidadeInput) cidadeInput.value = '';
            if (ufInput) ufInput.value = '';
        }
